% Field patches Paperpile will never carry (replaces bib_patches/patch_bib).
%
% maxnames=6 truncates long author lists to "et al", which hides the own name
% on large collaborations. Vidyasagaran2024-wx has 61 authors with Del Castillo
% at position 26, so the addendum states the contribution explicitly. Add one
% \map block per affected cite key — only needed when the own name sits beyond
% position 6.
%
% The name is bolded literally here: addendum is a plain field, so the
% \mkbibnamefamily hook below never sees it.
%
% Journal abbreviations need no patch: Paperpile exports them already
% abbreviated ("Int. J. Epidemiol."), and vancouver.bbx's own sourcemap then
% strips every period from journaltitle and appends a single terminal one,
% giving strict ICMJE/NLM form ("Int J Epidemiol."). Do not add a map to
% restore the internal periods — that is only how cv-professional looks
% because its .bib writes `journal` rather than `journaltitle`, and biber
% sourcemaps match the source field name before the BibTeX alias applies, so
% vancouver's map silently skips it there.
\DeclareSourcemap{%
  \maps[datatype=bibtex]{%
    \map{%
      \step[fieldsource=entrykey, match={Vidyasagaran2024-wx}, final]
      \step[fieldset=addendum,
            fieldvalue={Among authors: \textbf{Del Castillo Fernandez, D.}}]%
    }%
    % 13 authors, own name 8th, so maxnames=6 truncates it away. Byline form
    % here is "Del Castillo, Darwin" (not "...Fernandez"), so the addendum
    % mirrors this paper's byline rather than the Vidyasagaran one — the point
    % is that a reader can match the string against the published author list.
    \map{%
      \step[fieldsource=entrykey, match={Maslova2026-gt}, final]
      \step[fieldset=addendum,
            fieldvalue={Among authors: \textbf{Del Castillo, D.}}]%
    }%
    % Conference abstract (SSM/ECE 2026), now an @article in Paperpile carrying
    % only journaltitle + date. The typo, duplicate eventtitle and journal-as-
    % editor were all fixed upstream, so only one patch is left: nothing in the
    % entry marks this as a conference abstract rather than a full paper, and
    % no bib field carries presentation format.
    %
    % Use addendum, not note: on @article, note prints between volume and
    % pages ("80. Rapid-fire oral presentation.:A12-A13"), while addendum
    % prints last. On @inproceedings the two behave alike — this ordering
    % difference is entry-type-specific.
    %
    % Drop "; accepted" after the conference (Sept 2026). When the JECH
    % supplement assigns volume/pages/DOI, Paperpile may re-key the entry
    % again — the match below and the \cvpub in content/publications.tex both
    % silently stop resolving, so check build/cv.blg after any sync.
    \map{%
      \step[fieldsource=entrykey, match={Del-Castillo2026-uy}, final]
      \step[fieldset=addendum,
            fieldvalue={Rapid-fire oral presentation, 70th Society for Social Medicine \& Population Health Conference -- European Congress of Epidemiology 2026, London, UK; accepted.}]%
    }%
    % Conference presentations, NLM form:
    %   Author. Title. Presented at: Event; Year Mon Day; City, Country.
    % Two things stop vancouver.bbx producing that on its own:
    %   - its @misc driver reads neither eventtitle nor venue, so the event
    %     vanishes; addendum is the field it does print (see Vidyasagaran above)
    %   - NLM puts the date mid-string, next to the event, but vancouver prints
    %     date at the end of the entry. So the date is written into addendum by
    %     hand and the real field nulled, or the year prints twice. Keep the
    %     addendum date in sync if Paperpile revises the entry's date.
    % "and" not "&": a literal ampersand is an alignment tab, and \& would have
    % to survive biblatex serialising the fieldvalue into build/cv.bcf.
    % The title's "Rapid-Fire Presentation:" prefix is Paperpile's; stripped so
    % the format marker sits where NLM puts it, next to the event.
    \map{%
      \step[fieldsource=entrykey, match={Del-Castillo2026-jq}, final]
      \step[fieldsource=title,
            match=\regexp{Rapid-Fire\s+Presentation:\s*}, replace={}]
      \step[fieldset=addendum,
            fieldvalue={Rapid-fire presentation at: 70th Society for Social
                        Medicine and Population Health Conference -- European
                        Congress of Epidemiology; 2026 Sep 11; London, UK
                        \textit{(accepted, forthcoming)}}]%
      \step[fieldset=date, null]%
    }%
  }%
}

% Bold own name (replaces filters/bold-author.lua). Same rule: family name
% beginning "Del Castillo" (covers "Del Castillo", "Del Castillo Fernández",
% "Del Castillo-Fernández") anchored on a given name starting with "D".
\usepackage{xstring}
\newif\ifownname
\newcommand*{\checkownname}{%
  \global\ownnamefalse
  \ifdefvoid{\namepartgiven}{}{%
    \ifdefvoid{\namepartfamily}{}{%
      \begingroup\expandarg
        % \namepartfamily is 'Del\bibnamedelima Castillo' — delimiter macro
        % between tokens, so prefix-match on 'Del Castillo' can never hit.
        % Substring tests skip the delimiter: family must contain both
        % 'Del' and 'Castillo', given name must start with 'D'.
        \IfSubStr{\namepartfamily}{Castillo}
          {\IfSubStr{\namepartfamily}{Del}
            {\IfBeginWith{\namepartgiven}{D}{\global\ownnametrue}{}}
            {}}%
          {}%
      \endgroup
    }%
  }%
}
\renewcommand*{\mkbibnamefamily}[1]{\checkownname\ifownname\textbf{#1}\else#1\fi}
\renewcommand*{\mkbibnamegiven}[1]{\checkownname\ifownname\textbf{#1}\else#1\fi}
