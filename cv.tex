% \cventry{<text>}{<date>}: date flush right on the first line, later lines hang by 1em.
% Use \newline, not \\: under \raggedright, \\ ends the paragraph and the next line loses its hang.
\newlength{\cvdatewidth}
\newenvironment{cvledger}{\par\nobreak\setlength{\parskip}{0.3\baselineskip}}{\par}
\newcommand{\cventry}[2]{%
  \settowidth{\cvdatewidth}{#2}%
  {\raggedright\interlinepenalty=10000
   \parshape 2
     0pt \dimexpr\linewidth-\cvdatewidth-1.25em\relax
     1em \dimexpr\linewidth-1em\relax
   \noindent\rlap{\hspace*{\dimexpr\linewidth-\cvdatewidth\relax}#2}#1\par}}

% Per-entry patches live here, not in the .bib, because Paperpile overwrites the .bib on every sync.
% Journal abbreviations need none: vancouver.bbx already strips their periods to NLM form.
\DeclareSourcemap{%
  \maps[datatype=bibtex]{%
    % maxnames=6 truncates the own name away on these papers, so the addendum states it,
    % spelled as each paper's byline does.
    \map{%
      \step[fieldsource=entrykey, match={Vidyasagaran2024-wx}, final]
      \step[fieldset=addendum,
            fieldvalue={Among authors: \textbf{Del Castillo Fernandez, D.}}]%
    }%
    \map{%
      \step[fieldsource=entrykey, match={Maslova2026-gt}, final]
      \step[fieldset=addendum,
            fieldvalue={Among authors: \textbf{Del Castillo, D.}}]%
    }%
    % vancouver's @misc driver prints neither eventtitle nor venue, so the event goes in addendum.
    % The date is typed here and the date field nulled (NLM puts it mid-string); keep it in sync with Paperpile.
    % "and", not "&": a bare & is an alignment tab and \& does not survive the trip through cv.bcf.
    \map{%
      \step[fieldsource=entrykey, match={Del-Castillo2026-jq}, final]
      \step[fieldsource=title,
            match=\regexp{Rapid-Fire\s+Presentation:\s*}, replace={}]
      \step[fieldset=addendum,
            fieldvalue={Rapid-fire presentation at: 70th Society for Social
                        Medicine and Population Health Conference -- European
                        Congress of Epidemiology; 2026 Sep 11; London, UK
                        }]%
      \step[fieldset=date, null]%
    }%
  }%
}
